**Goal:** friends can find and join each other's games through Steam (relay, lobbies, invites) with server-verified identity, from a Linux or Windows build installed from a Steam Playtest branch.

**Architecture:** Steam-first, KCP kept as "Direct". Host listens on Mirror's multiplexer (KCP plus FizzySteamworks); a client picks its transport from a `ConnectionRoute`. Everything Steam-specific sits behind `ISteamBackend` (pure interface and logic in the `WAR2D` assembly, tested with a fake) with the real Steamworks.NET implementation in a separate `WAR2D.Steam` assembly guarded by the `WAR2D_STEAM` define. A Mirror `NetworkAuthenticator` verifies Steam auth tickets.

**Tech Stack:** Unity `6000.4.1f1`, Mirror (KCP + Multiplexer), Steamworks.NET, FizzySteamworks (MIT), `steamcmd` / SteamPipe, UI Toolkit.

```csharp
namespace WAR2D.Net.Steam
{
    /// <summary>How a client reaches a host.</summary>
    public enum RouteKind : byte { Direct, Steam }

    /// <summary>Where a client connects: an address and port (KCP), or a host's Steam ID (Steam relay).</summary>
    public readonly struct ConnectionRoute
    {
        public RouteKind Kind { get; }
        public string Host { get; }
        public ushort Port { get; }
        public ulong SteamId { get; }

        private ConnectionRoute(RouteKind kind, string host, ushort port, ulong steamId)
        {
            Kind = kind; Host = host; Port = port; SteamId = steamId;
        }

        public static ConnectionRoute ForAddress(string host, ushort port) => new ConnectionRoute(RouteKind.Direct, host, port, 0);
        public static ConnectionRoute ForSteam(ulong steamId) => new ConnectionRoute(RouteKind.Steam, null, 0, steamId);

        /// <summary><c>steam:&lt;id&gt;</c> is a Steam route; anything else must be a valid <see cref="JoinAddress"/>.</summary>
        public static bool TryParse(string text, ushort defaultPort, out ConnectionRoute route)
        {
            route = default;
            if (string.IsNullOrWhiteSpace(text)) return false;
            text = text.Trim();
            if (text.StartsWith("steam:", System.StringComparison.OrdinalIgnoreCase))
            {
                string digits = text.Substring(6);
                if (digits.Length == 0 || digits[0] == '+' || digits[0] == '-') return false;
                if (!ulong.TryParse(digits, out ulong id) || id == 0) return false;
                route = ForSteam(id);
                return true;
            }
            if (!JoinAddress.TryParse(text, defaultPort, out string host, out ushort port)) return false;
            route = ForAddress(host, port);
            return true;
        }

        public override string ToString() => Kind == RouteKind.Steam ? $"steam:{SteamId}" : $"{Host}:{Port}";
    }
}
```

```csharp
using System;
using System.Collections.Generic;

namespace WAR2D.Net.Steam
{
    /// <summary>One lobby as the browser shows it, and the metadata keys the host publishes.</summary>
    public struct LobbyInfo : IEquatable<LobbyInfo>
    {
        public const string KeyVersion = "ver", KeyState = "state", KeyMode = "mode", KeyMapSize = "size", KeyHostName = "host";

        public ulong LobbyId, HostSteamId;
        public string HostName, Version, State, Mode;
        public int MapSize, Players;

        /// <summary>The key/value pairs the host sets on the Steam lobby.</summary>
        public Dictionary<string, string> Encode() => new Dictionary<string, string>
        {
            [KeyVersion] = Version, [KeyState] = State, [KeyMode] = Mode,
            [KeyMapSize] = MapSize.ToString(System.Globalization.CultureInfo.InvariantCulture), [KeyHostName] = HostName,
        };

        /// <summary>Reads lobby data a stranger published. Anything missing or malformed is rejected; the name is sanitised.</summary>
        public static bool TryDecode(ulong lobbyId, ulong hostSteamId, int players, IReadOnlyDictionary<string, string> data, out LobbyInfo info)
        {
            info = default;
            if (data == null) return false;
            if (!data.TryGetValue(KeyVersion, out string ver) || string.IsNullOrEmpty(ver)) return false;
            if (!data.TryGetValue(KeyState, out string state) || string.IsNullOrEmpty(state)) return false;
            if (!data.TryGetValue(KeyMode, out string mode) || string.IsNullOrEmpty(mode)) return false;
            if (!data.TryGetValue(KeyMapSize, out string sizeText)
                || !int.TryParse(sizeText, System.Globalization.NumberStyles.None, System.Globalization.CultureInfo.InvariantCulture, out int size)) return false;
            if (!data.TryGetValue(KeyHostName, out string rawName) || !CommandValidator.TrySanitizeNickname(rawName, out string name)) return false;
            info = new LobbyInfo
            {
                LobbyId = lobbyId, HostSteamId = hostSteamId, HostName = name, Version = ver, State = state,
                Mode = mode, MapSize = size, Players = players,
            };
            return true;
        }

        public bool Equals(LobbyInfo o) =>
            LobbyId == o.LobbyId && HostSteamId == o.HostSteamId && HostName == o.HostName && Version == o.Version
            && State == o.State && Mode == o.Mode && MapSize == o.MapSize && Players == o.Players;
        public override bool Equals(object obj) => obj is LobbyInfo o && Equals(o);
        public override int GetHashCode() => LobbyId.GetHashCode();
    }
}
```

```csharp
using System.Collections.Generic;

namespace WAR2D.Net.Steam
{
    /// <summary>Which lobbies the browser lists: open, not full, same build; busiest first.</summary>
    public static class LobbyListModel
    {
        public static List<LobbyInfo> Visible(IEnumerable<LobbyInfo> all, string myVersion)
        {
            var shown = new List<LobbyInfo>();
            foreach (LobbyInfo l in all)
                if (l.State == "lobby" && l.Players < LobbyRules.MaxPlayers && GameVersion.IsCompatible(l.Version, myVersion)) shown.Add(l);
            shown.Sort((a, b) => a.Players != b.Players ? b.Players.CompareTo(a.Players) : string.CompareOrdinal(a.HostName, b.HostName));
            return shown;
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;
using Mirror;

namespace WAR2D.Net.Steam
{
    public struct SteamAuthRequest : NetworkMessage { public string version; public ulong steamId; public byte[] ticket; }
    public struct SteamAuthResponse : NetworkMessage { public bool accepted; public string reason; }

    /// <summary>The rules a Steam sign-in must pass before Steam is even asked. Pure.</summary>
    public static class SteamAuthPolicy
    {
        public const int MaxTicketBytes = 1024;

        public static bool Check(string hostVersion, SteamAuthRequest req, ulong transportSteamId, out string reason)
        {
            if (!GameVersion.IsCompatible(hostVersion, req.version)) { reason = $"Wrong game version (host {hostVersion})."; return false; }
            if (transportSteamId == 0 || req.steamId != transportSteamId) { reason = "Steam identity does not match the connection."; return false; }
            if (req.ticket == null || req.ticket.Length == 0 || req.ticket.Length > MaxTicketBytes) { reason = "Bad Steam ticket."; return false; }
            reason = null;
            return true;
        }

        /// <summary>Direct (KCP, no Steam sign-in) connections: the editor, <c>-perf</c> runs, or <c>-direct</c>.</summary>
        public static bool DirectAllowed(bool isEditor, bool devApi, string[] commandLine) =>
            isEditor || devApi || Array.IndexOf(commandLine ?? new string[0], "-direct") >= 0;
    }

    /// <summary>Server-side bookkeeping of sign-ins waiting for Steam's verdict. Pure apart from the backend calls.</summary>
    public sealed class SteamAuthSessions
    {
        public enum Begin { Rejected, Pending }

        private readonly ISteamBackend backend;
        private readonly string hostVersion;
        private readonly double timeout;
        private readonly Dictionary<ulong, (int conn, double deadline)> pending = new Dictionary<ulong, (int, double)>();
        private readonly List<(int, bool, string)> finished = new List<(int, bool, string)>();

        public SteamAuthSessions(ISteamBackend backend, string hostVersion, double timeoutSeconds = 10)
        {
            this.backend = backend;
            this.hostVersion = hostVersion;
            timeout = timeoutSeconds;
            backend.AuthSessionResult += OnResult;
        }

        public Begin OnRequest(int connectionId, SteamAuthRequest req, ulong transportSteamId, double now, out string reason)
        {
            if (!SteamAuthPolicy.Check(hostVersion, req, transportSteamId, out reason)) return Begin.Rejected;
            if (pending.ContainsKey(req.steamId)) { reason = "Already signing in."; return Begin.Rejected; }
            if (!backend.BeginAuthSession(req.steamId, req.ticket)) { reason = "Steam refused the ticket."; return Begin.Rejected; }
            pending[req.steamId] = (connectionId, now + timeout);
            return Begin.Pending;
        }

        private void OnResult(ulong steamId, bool accepted)
        {
            if (!pending.TryGetValue(steamId, out var p)) return;
            pending.Remove(steamId);
            if (!accepted) backend.EndAuthSession(steamId);
            finished.Add((p.conn, accepted, accepted ? null : "Steam could not verify you."));
        }

        /// <summary>Verdicts since the last call; also times out sign-ins Steam never answered.</summary>
        public List<(int connectionId, bool accepted, string reason)> Drain(double now)
        {
            List<ulong> late = null;
            foreach (var kv in pending) if (now >= kv.Value.deadline) (late ??= new List<ulong>()).Add(kv.Key);
            if (late != null)
                foreach (ulong id in late)
                {
                    finished.Add((pending[id].conn, false, "Steam did not answer in time."));
                    pending.Remove(id);
                    backend.EndAuthSession(id);
                }
            var result = new List<(int, bool, string)>(finished);
            finished.Clear();
            return result;
        }

        /// <summary>Drops a connection's pending sign-in (it disconnected).</summary>
        public void Forget(int connectionId)
        {
            ulong key = 0; bool found = false;
            foreach (var kv in pending) if (kv.Value.conn == connectionId) { key = kv.Key; found = true; break; }
            if (!found) return;
            pending.Remove(key);
            backend.EndAuthSession(key);
        }

        public void Dispose() => backend.AuthSessionResult -= OnResult;
    }

    /// <summary>Server-side map from connection to verified Steam ID. Never sent to clients.</summary>
    public static class SteamIdentity
    {
        private static readonly Dictionary<int, ulong> Ids = new Dictionary<int, ulong>();
        public static void Set(int connectionId, ulong steamId) => Ids[connectionId] = steamId;
        public static bool TryGet(int connectionId, out ulong steamId) => Ids.TryGetValue(connectionId, out steamId);
        public static void Forget(int connectionId) => Ids.Remove(connectionId);
        public static void Clear() => Ids.Clear();
    }
```

```csharp
namespace WAR2D.Net.Steam
{
    /// <summary>Keeps the host's Steam lobby listing in step with the game lobby.</summary>
    public sealed class SteamLobbyService
    {
        private readonly ISteamBackend backend;
        private string hostName;
        private LobbyInfo last;
        private bool open;

        public SteamLobbyService(ISteamBackend backend) { this.backend = backend; }

        public ulong LobbyId { get; private set; }

        public void Open(string hostName, MatchSettings settings, int players)
        {
            this.hostName = hostName;
            backend.CreateLobby(LobbyRules.MaxPlayers, id =>
            {
                if (id == 0) { UnityEngine.Debug.LogWarning("[Steam] could not create a lobby."); return; }
                LobbyId = id;
                open = true;
                last = default;
                Push(GameState.Lobby, settings, players);
            });
        }

        public void Update(GameState state, MatchSettings settings, int players)
        {
            if (open) Push(state, settings, players);
        }

        private void Push(GameState state, MatchSettings settings, int players)
        {
            bool joinable = state == GameState.Lobby;
            LobbyInfo now = new LobbyInfo
            {
                LobbyId = LobbyId, HostSteamId = backend.LocalSteamId, HostName = hostName, Version = GameVersion.Current,
                State = joinable ? "lobby" : "playing", Mode = settings.Mode.ToString(), MapSize = settings.MapSize, Players = players,
            };
            if (now.Equals(last)) return;
            bool firstPush = last.LobbyId == 0;
            if (firstPush || last.State != now.State) backend.SetLobbyJoinable(LobbyId, joinable);
            backend.SetLobbyData(LobbyId, now.Encode());
            backend.SetRichPresence(joinable ? "In a lobby" : "In a match");
            last = now;
        }

        public void Close()
        {
            if (!open) return;
            open = false;
            backend.LeaveLobby(LobbyId);
            backend.SetRichPresence(null);
        }
    }
}
```

```csharp
        private Callback<GameLobbyJoinRequested_t> joinRequested;

        private CallResult<LobbyEnter_t> entered;

        {

            if (!initialised) return;
```

```csharp
using System.Collections.Generic;
using Mirror;
using UnityEngine;

namespace WAR2D.Net.Steam
{
    /// <summary>
    /// Authenticates Steam relay connections with Steam auth tickets, and lets Direct (KCP) connections
    /// through only where <see cref="SteamAuthPolicy.DirectAllowed"/> says so.
    /// </summary>
    public class SteamAuthenticator : NetworkAuthenticator
    {
        private SteamAuthSessions sessions;
        private readonly Dictionary<int, NetworkConnectionToClient> waiting = new Dictionary<int, NetworkConnectionToClient>();

        public override void OnStartServer()
        {
            NetworkServer.RegisterHandler<SteamAuthRequest>(OnRequest, false);
            if (SteamServices.Available) sessions = new SteamAuthSessions(SteamServices.Backend, GameVersion.Current);
        }

        public override void OnStopServer()
        {
            NetworkServer.UnregisterHandler<SteamAuthRequest>();
            sessions?.Dispose();
            sessions = null;
            waiting.Clear();
        }

        public override void OnServerAuthenticate(NetworkConnectionToClient conn)
        {
            if (TransportSteamId(conn.connectionId) == 0)
            {
                // A KCP connection: no Steam identity to verify.
                if (SteamAuthPolicy.DirectAllowed(Application.isEditor, DevApi.Allowed, System.Environment.GetCommandLineArgs())) ServerAccept(conn);
                else Reject(conn, "Direct connections are off on this host.");
                return;
            }
            waiting[conn.connectionId] = conn; // the client's request is on its way; Update() enforces the timeout
        }

        private void OnRequest(NetworkConnectionToClient conn, SteamAuthRequest msg)
        {
            if (sessions == null) { Reject(conn, "This host is not signed in to Steam."); return; }
            ulong claimed = TransportSteamId(conn.connectionId);
            if (sessions.OnRequest(conn.connectionId, msg, claimed, NetworkTime.localTime, out string reason) == SteamAuthSessions.Begin.Rejected)
                Reject(conn, reason);
        }

        private void Update()
        {
            if (sessions == null) return;
            foreach (var (connectionId, accepted, reason) in sessions.Drain(NetworkTime.localTime))
            {
                if (!waiting.TryGetValue(connectionId, out NetworkConnectionToClient conn)) continue;
                waiting.Remove(connectionId);
                if (!accepted) { Reject(conn, reason); continue; }
                SteamIdentity.Set(connectionId, TransportSteamId(connectionId));
                conn.Send(new SteamAuthResponse { accepted = true });
                ServerAccept(conn);
            }
        }

        private void Reject(NetworkConnectionToClient conn, string reason)
        {
            Debug.LogWarning($"[Steam] rejecting connection {conn.connectionId}: {reason}");
            waiting.Remove(conn.connectionId);
            sessions?.Forget(conn.connectionId);
            conn.Send(new SteamAuthResponse { accepted = false, reason = reason });
            ServerReject(conn);
        }

        /// <summary>The Steam ID behind a connection, or 0 when it is not a Steam relay connection.</summary>
        private static ulong TransportSteamId(int connectionId) =>
            ulong.TryParse(Transport.active?.ServerGetClientAddress(connectionId), out ulong id) ? id : 0;

        public override void OnStartClient() => NetworkClient.RegisterHandler<SteamAuthResponse>(OnResponse, false);
        public override void OnStopClient() => NetworkClient.UnregisterHandler<SteamAuthResponse>();

        public override void OnClientAuthenticate()
        {
            if (GameManager.Instance.ActiveRoute.Kind != RouteKind.Steam) { ClientAccept(); return; }
            if (!SteamServices.Available || !SteamServices.Backend.TryGetAuthTicket(out byte[] ticket)) { ClientReject(); return; }
            NetworkClient.Send(new SteamAuthRequest { version = GameVersion.Current, steamId = SteamServices.Backend.LocalSteamId, ticket = ticket });
        }

        private void OnResponse(SteamAuthResponse msg)
        {
            if (msg.accepted) ClientAccept();
            else { Debug.LogWarning($"[Steam] the host refused you: {msg.reason}"); ClientReject(); }
        }
    }
}
```

```csharp
using System.IO;
using Mirror;
using UnityEngine;

/// <summary>Writes the bytes this process receives and sends through Mirror each second (relay bandwidth check).</summary>
public sealed class RelayMeter : MonoBehaviour
{
    private long inBytes, outBytes;
    private float next;
    private StreamWriter file;

    [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)]
    private static void Boot()
    {
        string[] args = System.Environment.GetCommandLineArgs();
        int i = System.Array.IndexOf(args, "-relayMeter");
        if (i < 0 || i + 1 >= args.Length || !DevApi.Allowed) return;
        new GameObject("RelayMeter").AddComponent<RelayMeter>().Begin(args[i + 1]);
    }

    private void Begin(string path)
    {
        DontDestroyOnLoad(gameObject);
        file = new StreamWriter(path) { AutoFlush = true };
        file.WriteLine("t,inBytes,outBytes");
        NetworkDiagnostics.InMessageEvent += m => System.Threading.Interlocked.Add(ref inBytes, m.bytes * m.count);
        NetworkDiagnostics.OutMessageEvent += m => System.Threading.Interlocked.Add(ref outBytes, m.bytes * m.count);
        next = Time.unscaledTime + 1f;
    }

    private void Update()
    {
        if (file == null || Time.unscaledTime < next) return;
        next += 1f;
        file.WriteLine($"{Time.unscaledTime:F0},{System.Threading.Interlocked.Exchange(ref inBytes, 0)},{System.Threading.Interlocked.Exchange(ref outBytes, 0)}");
    }

    private void OnDestroy() => file?.Dispose();
}
```

```csharp
    /// <summary>Release builds for the Steam depots.</summary>
    public static class BuildCommands
    {
        [MenuItem("WAR-2D/Build/Linux (Steam)")] public static void Linux() => Build(BuildTarget.StandaloneLinux64, "Builds/Steam/linux/WAR-2D.x86_64");
        [MenuItem("WAR-2D/Build/Windows (Steam)")] public static void Windows() => Build(BuildTarget.StandaloneWindows64, "Builds/Steam/windows/WAR-2D.exe");

        private static void Build(BuildTarget target, string path)
        {
            var scenes = new List<string>();
            foreach (var s in EditorBuildSettings.scenes) if (s.enabled) scenes.Add(s.path);
            BuildReport report = BuildPipeline.BuildPlayer(new BuildPlayerOptions
            {
                scenes = scenes.ToArray(), locationPathName = path, target = target, options = BuildOptions.None,
            });
            if (report.summary.result != BuildResult.Succeeded) throw new System.Exception($"Build failed: {report.summary.result}");
            UnityEngine.Debug.Log($"Built {target} to {path} ({report.summary.totalSize / 1048576} MB)");
        }
    }
}
```

```bash
#!/usr/bin/env bash
# Usage: tools/steam-upload.sh <branch> [description]
# Uploads Builds/Steam/linux and Builds/Steam/windows to the given Steam branch through steamcmd.
# Needs steam/steam.env (see steam/steam.env.example). steamcmd asks for the password and Steam Guard

branch="${1:?usage: tools/steam-upload.sh <branch> [description]}"
desc="${2:-v$(grep -m1 'bundleVersion:' ProjectSettings/ProjectSettings.asset | awk '{print $2}') $(git rev-parse --short HEAD)}"

source steam/steam.env
for v in APP_ID DEPOT_LINUX DEPOT_WINDOWS STEAM_USER; do [[ -n "${!v:-}" ]] || { echo "$v is empty in steam/steam.env"; exit 1; }; done
for d in Builds/Steam/linux Builds/Steam/windows; do [[ -d "$d" ]] || { echo "Missing $d: build it first (WAR-2D/Build menu)"; exit 1; }; done

out="$(realpath -m steam/output)"; mkdir -p "$out"
cat > "$out/depot_linux.vdf" <<EOF
"DepotBuildConfig" { "DepotID" "$DEPOT_LINUX" "ContentRoot" "$(realpath Builds/Steam/linux)" "FileMapping" { "LocalPath" "*" "DepotPath" "." "recursive" "1" } "FileExclusion" "steam_appid.txt" }
EOF
cat > "$out/depot_windows.vdf" <<EOF
"DepotBuildConfig" { "DepotID" "$DEPOT_WINDOWS" "ContentRoot" "$(realpath Builds/Steam/windows)" "FileMapping" { "LocalPath" "*" "DepotPath" "." "recursive" "1" } "FileExclusion" "steam_appid.txt" }
EOF
cat > "$out/app_build.vdf" <<EOF
"AppBuild" { "AppID" "$APP_ID" "Desc" "$desc" "BuildOutput" "$out/logs" "SetLive" "$branch" "Depots" { "$DEPOT_LINUX" "$out/depot_linux.vdf" "$DEPOT_WINDOWS" "$out/depot_windows.vdf" } }
EOF
"$STEAMCMD" +login "$STEAM_USER" +run_app_build "$out/app_build.vdf" +quit
```
